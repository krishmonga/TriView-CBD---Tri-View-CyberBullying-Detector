% Stage 0
\node[stg=trifuseBlue, minimum width=10.9cm, text width=10.6cm] (raw) at (5.45, 0) {%

  Dataset~I (Shahane, 229,228) \textbar\ Dataset~II (Davidson, 24,783)};

% Stages 1–5 (numbered badges + boxes)
\node[badge=trifuseBlue] (n1) at (0.35, -1.35) {1};
\node[stg=trifuseBlue] (s1) at (5.45, -1.35) {%
  \textbf{Text cleaning \& normalization}\quad
  lowercase; URL$\rightarrow$\texttt{[URL]}; @$\rightarrow$\texttt{[USER]}; emoticons;
  char-trim; keep !?.,; collapse spaces};

\node[badge=trifuseBlue] (n2) at (0.35, -2.65) {2};
\node[stg=trifuseBlue] (s2) at (5.45, -2.65) {%
  \textbf{NLTK word tokenization}\quad $X=\{w_1,w_2,\ldots,w_n\}$};

\node[badge=trifuseBlue] (n3) at (0.35, -3.95) {3};
\node[stg=trifuseBlue] (s3) at (5.45, -3.95) {%
  \textbf{Vocabulary \& integer coding}\quad
  top 20,000 tokens; OOV$\rightarrow$\texttt{<UNK>}; special tokens
  \texttt{<PAD>}, \texttt{<BOS>}, \texttt{<EOS>}};

\node[badge=trifuseBlue] (n4) at (0.35, -5.25) {4};
\node[stg=trifuseBlue] (s4) at (5.45, -5.25) {%
  \textbf{Sequence standardization}\quad pad or truncate to $L=128$};

\node[badge=trifuseBlue] (n5) at (0.35, -6.55) {5};
\node[stg=trifuseBlue] (s5) at (5.45, -6.55) {%
  \textbf{Embedding tensor}\quad $E\in\mathbb{R}^{128\times 300}$};

\node[path=trifuseOrange] (glove) at (2.85, -7.95) {%
  \textbf{Baselines}\\
  GloVe 300-d vectors};
\node[path=trifuseGreen] (bert) at (8.05, -7.95) {%
  \textbf{TriFuse}\\
  DistilBERT partial FT + MLP};

% Output
\node[rectangle, rounded corners=4pt, draw=trifuseRed!60, fill=trifuseRed!8, thick,

  \textbf{Encoded input} ready for lexical, semantic, and structural branches};
